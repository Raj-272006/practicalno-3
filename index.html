<?php
function sumN($n) {
    $sum = 0;
    for($i = 1; $i <= $n; $i++) $sum += $i;
    return $sum;
}
function factorial($n) {
    $fact = 1;
    for($i = 1; $i <= $n; $i++) $fact *= $i;
    return $fact;
}
if(isset($_POST['submit'])) {
    $n = $_POST['n'];
    echo "Sum of first $n numbers = " . sumN($n) . "<br>";
    echo "Factorial of $n = " . factorial($n);

Q4. Length Without Built-in Function and Count Vowels

<?php
function stringLength($str) {
    $count=0;
    while(isset($str[$count])) $count++;
    return $count;
}
function countVowels($str) {
    $count=0; $str=strtolower($str);
    for($i=0;$i<strlen($str);$i++)
        if(in_array($str[$i],['a','e','i','o','u'])) $count++;
    return $count;
}
$str=$_POST['str'] ?? "Hello World";

Q1. Student-Competition Database

-- SQL tables
CREATE DATABASE college;
USE college;

CREATE TABLE Student(
  Stud_id INT PRIMARY KEY,
  Name VARCHAR(50),
  Class VARCHAR(20)
);

CREATE TABLE Competition(
  C_no INT PRIMARY KEY,
  C_name VARCHAR(50),
  Type VARCHAR(30)
);

CREATE TABLE Student_Competition(
  Stud_id INT,
  C_no INT,
  Rank_no INT,
  Year INT,
  PRIMARY KEY(Stud_id,C_no,Year),
  FOREIGN KEY(Stud_id) REFERENCES Student(Stud_id),
  FOREIGN KEY(C_no) REFERENCES Competition(C_no)
);

-- PHP query for 1st rank
<?php

$competition=$_POST['competition'] ?? '';

FROM Student JOIN Student_Competition ON Student.Stud_id=Student_Competition.Stud_id
JOIN Competition ON Competition.C_no=Student_Competition.C_no
WHERE Competition.C_name='$competition' AND Student_Competition.Rank_no=1";
$result=mysqli_query($conn,$sql);

    var password=document.getElementById("password").value;

    if(username.length<6){alert("Username must contain at least 6 characters.");return false;}
